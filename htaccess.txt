# Cache-Control: immutable dice al browser di non rivalidare nemmeno al ricarico
# della pagina. Applicarlo a tutto sarebbe rischioso: un file modificato senza
# cambiare URL resterebbe quello vecchio per un anno.
#
# Viene quindi applicato solo agli asset con "ver=" nella query string, che e'
# esattamente il caso di cio' che WordPress accoda: l'URL cambia quando cambia il
# contenuto, quindi non rivalidare non costa nulla.
<IfModule mod_setenvif.c>
    SetEnvIf Query_String "(^|&)ver=" SPEEDUP_VERSIONED_ASSET
</IfModule>

<IfModule mod_headers.c>
    Header set Cache-Control "public, max-age=31536000, immutable" env=SPEEDUP_VERSIONED_ASSET
</IfModule>
